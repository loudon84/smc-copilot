11. Startup Decision 重写

删除当前：

startup-decision.ts 应只依赖：

伪代码：

export async function resolveStartupDecision() {
  if (!hasAuth()) {
    return {
      nextScreen: "login",
      reason: "auth-required"
    };
  }

  if (!isUserBootstrapReady()) {
    return {
      nextScreen: "login",
      reason: "bootstrap-pending"
    };
  }

  const runtime = await bootCoordinator.runtime();

  switch (runtime.state) {
    case "Ready":
      return {
        nextScreen: "main",
        reason: "runtime-ready"
      };

    case "RuntimeDegraded":
      return {
        nextScreen: "main",
        reason: "runtime-degraded"
      };

    case "PairingRequired":
    case "RuntimeMissing":
    case "Incompatible":
      return {
        nextScreen: "runtime-recovery",
        reason: runtime.state
      };

    case "Connecting":
    case "RuntimeStarting":
      return {
        nextScreen: "runtime-recovery",
        reason: "runtime-starting"
      };
  }
}
